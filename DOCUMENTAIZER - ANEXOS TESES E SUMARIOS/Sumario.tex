% CONFIGURAÇÃO POR EMPRESA
% - Cada empresa tem um bloco dedicado de layout.
% - O gerador substitui <<EMPRESA>> por RUDA, HLA ou ABEL.

\newcommand{\EmpresaSelecionada}{<<EMPRESA>>}
\newcommand{\AplicarHeaderEmpresa}{}
\newcommand{\TituloSumarioFonte}{\sffamily\bfseries\Large}
\newcommand{\TituloSumarioCor}{black}

% ---------- RUDA ----------
\newcommand{\ConfigEmpresaRUDA}{%
  \renewcommand{\AplicarHeaderEmpresa}{%
    \AddToShipoutPictureBG{%
      \AtPageUpperLeft{%
        \hspace*{0.5\paperwidth}%
        \raisebox{-3cm}{\makebox[0pt][c]{\includegraphics[width=0.8\paperwidth]{Ruda_Header.png}}}%
      }%
    }%
  }%
  \renewcommand{\TituloSumarioFonte}{\sffamily\bfseries\Large}%
  \renewcommand{\TituloSumarioCor}{black}%
}

% ---------- ABEL ----------
\newcommand{\ConfigEmpresaABEL}{%
  \renewcommand{\AplicarHeaderEmpresa}{%
    \AddToShipoutPictureBG{\AtPageUpperLeft{\raisebox{-\height}{\includegraphics{ABEL_Header.png}}}}%
  }%
  \renewcommand{\TituloSumarioFonte}{\sffamily\bfseries\Large}%
  \renewcommand{\TituloSumarioCor}{black}%
}

% ---------- HLA (padrão/fallback) ----------
\newcommand{\ConfigEmpresaHLA}{%
  \renewcommand{\AplicarHeaderEmpresa}{%
    \AddToShipoutPictureBG{\AtPageUpperLeft{\raisebox{-\height}{\includegraphics{HLA_Header.png}}}}%
  }%
  \renewcommand{\TituloSumarioFonte}{\sffamily\bfseries\Large}%
  \renewcommand{\TituloSumarioCor}{black}%
}

% Seleção da configuração ativa
\ifthenelse{\equal{\EmpresaSelecionada}{RUDA}}{%
  \ConfigEmpresaRUDA
}{%
  \ifthenelse{\equal{\EmpresaSelecionada}{ABEL}}{%
    \ConfigEmpresaABEL
  }{%
    \ConfigEmpresaHLA
  }%
}
